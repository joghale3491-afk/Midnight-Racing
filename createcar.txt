function createCar(color, isPlayer=false, bodyStyle=null, detailLevel="rival", modelId=null){
  const root=new THREE.Group();
  const visualRig=new THREE.Group();visualRig.name='car-visual-rig';root.add(visualRig);
  root.userData.visualRig=visualRig;root.userData.wheels=[];root.userData.frontWheels=[];root.userData.brakeLights=[];root.userData.motionPhase=Math.random()*Math.PI*2;root.userData.catalogModelId=modelId||profile.garage.carId;
  const selectedModelId=modelId||profile.garage.carId;
  const spec=isPlayer&&detailLevel!=='p2'?(CAR_CATALOG.find(c=>c.id===selectedModelId)||CAR_CATALOG[0]):null;
  const modelSpec=spec||(modelId?CAR_CATALOG.find(c=>c.id===modelId):null);
  const carAttachments=isPlayer&&detailLevel!=='p2'?getCarAttachmentState(selectedModelId).equipped:{};
  const style=modelSpec?.body||bodyStyle||1;
  const design=Number.isFinite(Number(modelSpec?.design))?Number(modelSpec.design):((Math.max(1,Number(style)||1)-1)*2)%8;
  const visualId=Number.isFinite(Number(modelSpec?.visualId))?Number(modelSpec.visualId):((design*23+(Number(style)||1)*7)%140);
  const familyScales=[[.94,.97],[1.15,1.03],[1.18,.91],[1.06,.96],[1.02,1.17],[.98,.91],[1.16,1.08],[1.02,.94]][design%8];
  const variantIndex=Math.max(0,Number(modelSpec?.variantIndex??(visualId%20)))%20;
  // Wider range of wheelbase and body proportions makes models visibly different at a glance.
  const individualWidth=.84+((visualId*7)%23)*.016,individualLength=.86+((visualId*11)%22)*.016;
  const widthScale=(style===3?0.94:style===4?1.08:style===2?1.02:1)*familyScales[0]*individualWidth;
  const lengthScale=(style===2?1.08:style===4?1.04:style===3?.96:1)*familyScales[1]*individualLength;
  const bodyMat=new THREE.MeshPhysicalMaterial({color,roughness:.22,metalness:.72,clearcoat:1,clearcoatRoughness:.11,envMapIntensity:1.05});
  const darkMat=makeMat(0x0b111b,.34,.55);
  const glassMat=new THREE.MeshPhysicalMaterial({color:0x102b3d,roughness:.12,metalness:.42,transparent:true,opacity:.88,clearcoat:1,clearcoatRoughness:.08});
  const trimMat=makeMat(0x111923,.28,.72);
  const tireMat=makeMat(0x080b10,.92,.02);
  const rimMat=makeMat(0x9eb7c9,.2,.95);
  const add=(geo,mat,pos,rot=null,scale=null)=>{const m=new THREE.Mesh(geo,mat);m.position.set(...pos);if(rot)m.rotation.set(...rot);if(scale)m.scale.set(...scale);m.castShadow=true;m.receiveShadow=true;visualRig.add(m);return m;};
  const planShape=(halfWidth,front,rear,profile=variantIndex,section='body')=>{
    const sh=new THREE.Shape();
    // Twenty different plan-view profiles: tapered hypercars, squared muscle cars,
    // short hatchbacks, long GTs, roadsters, touring saloons and more.
    const profiles=[
      [.12,.90,.88,.54],[.34,1.08,1.02,.78],[.035,.78,.82,.42],[.22,.98,.94,.72],
      [.20,.94,.84,.60],[.30,1.04,.92,.58],[.28,1.08,1.06,.86],[.08,.88,.86,.50],
      [.06,.84,.90,.46],[.26,1.02,.94,.72],[.24,1.05,1.04,.82],[.10,.96,.88,.52],
      [.02,.82,.86,.40],[.32,1.04,.95,.76],[.18,.98,1.02,.84],[.24,1.10,1.08,.88],
      [.04,.80,.84,.38],[.28,1.02,.94,.66],[.16,1.14,1.12,.90],[.08,.92,.92,.48]
    ][Math.abs(Number(profile)||0)%20];
    let [nose,frontShoulder,rearShoulder,tail]=profiles;
    if(section==='cabin'){nose=Math.min(.42,nose+.12);frontShoulder*=.70;rearShoulder*=.72;tail*=.78;}
    if(section==='roof'){nose=Math.min(.48,nose+.08);frontShoulder*=.58;rearShoulder*=.60;tail*=.68;}
    sh.moveTo(0,front);
    sh.quadraticCurveTo(halfWidth*nose,front-.10,halfWidth*frontShoulder,front-.58);
    sh.bezierCurveTo(halfWidth*(frontShoulder*1.08),front-1.12,halfWidth*(rearShoulder*1.05),rear+.88,halfWidth*rearShoulder,rear+.20);
    sh.quadraticCurveTo(halfWidth*tail,rear-.04,0,rear);
    sh.quadraticCurveTo(-halfWidth*tail,rear-.04,-halfWidth*rearShoulder,rear+.20);
    sh.bezierCurveTo(-halfWidth*(rearShoulder*1.05),rear+.88,-halfWidth*(frontShoulder*1.08),front-1.12,-halfWidth*frontShoulder,front-.58);
    sh.quadraticCurveTo(-halfWidth*nose,front-.10,0,front);
    return sh;
  };
  // Lightweight but still detailed enough AI model: the player cars keep the full sculpted model below,
  // while traffic and distant rivals use a compact 10-mesh LOD to protect frame rate and avoid GPU overload.
  if(!isPlayer){
    const simpleBody=add(new THREE.BoxGeometry(1.82*widthScale,.57,4.05*lengthScale),bodyMat,[0,.91,0]);
    simpleBody.material=bodyMat;
    const cabinHeight=[1.41,1.47,1.32,1.55,1.42,1.30,1.50,1.35][design%8]+(((visualId*3)%7)-3)*.014;
    const cabinLength=[1.74,1.55,1.82,1.90,1.56,1.50,1.82,1.72][design%8]*(.92+((visualId*5)%9)*.02);
    add(new THREE.BoxGeometry((design===1||design===6?1.46:1.34)*widthScale,.58,cabinLength*lengthScale),glassMat,[0,cabinHeight,-.08*lengthScale]);
    add(new THREE.BoxGeometry(1.92*widthScale,.13,.22),darkMat,[0,.66,2.0*lengthScale]);
    add(new THREE.BoxGeometry(1.9*widthScale,.11,.2),darkMat,[0,.67,-2.0*lengthScale]);
    const liteHead=new THREE.MeshStandardMaterial({color:0xeafaff,emissive:0x9beaff,emissiveIntensity:2.2,roughness:.2});
    const liteTail=new THREE.MeshStandardMaterial({color:0xff244f,emissive:0xff163d,emissiveIntensity:2,roughness:.24});
    for(const x of [-.65,.65]){
      add(new THREE.BoxGeometry(.34,.12,.08),liteHead,[x*widthScale,.98,2.04*lengthScale]);
      const tail=add(new THREE.BoxGeometry(.38,.12,.08),liteTail,[x*widthScale,.94,-2.04*lengthScale]);root.userData.brakeLights.push(tail);
    }
    const liteWheelGeo=new THREE.CylinderGeometry(.39,.39,.28,14,1,false);
    const liteRimGeo=new THREE.CylinderGeometry(.22,.22,.3,12,1,false);
    for(const x of [-.91,.91])for(const z of [-1.34,1.34]){
      const wheel=add(liteWheelGeo,tireMat,[x*widthScale,.43,z*lengthScale],[0,0,Math.PI/2]);
      root.userData.wheels.push(wheel);if(z>0)root.userData.frontWheels.push(wheel);
      const wheelFaceX=x*widthScale+(x>0?.155:-.155);
      add(liteRimGeo,rimMat,[wheelFaceX,.43,z*lengthScale],[0,0,Math.PI/2]);
      if(detailLevel!=='traffic'){
        add(new THREE.TorusGeometry(.205,.036,8,20),makeMat(0x263d50,.2,.95),[wheelFaceX,.43,z*lengthScale],[0,Math.PI/2,0]);
        add(new THREE.CylinderGeometry(.095,.095,.32,12),makeMat(0xe1edf5,.2,.95),[x*widthScale+(x>0?.17:-.17),.43,z*lengthScale],[0,0,Math.PI/2]);
      }
    }
    if(detailLevel!=='traffic'){
      const accentColor=new THREE.Color(color).offsetHSL(.075,-.18,.19).getHex();
      const accentMat=new THREE.MeshStandardMaterial({color:accentColor,roughness:.27,metalness:.76,emissive:accentColor,emissiveIntensity:.16});
      const glowMat=new THREE.MeshStandardMaterial({color:0xe8fbff,emissive:0x72eaff,emissiveIntensity:2.7,roughness:.18});
      const redGlow=new THREE.MeshStandardMaterial({color:0xff4268,emissive:0xff174c,emissiveIntensity:2.5,roughness:.22});
      const stripeMat=new THREE.MeshBasicMaterial({color:accentColor,transparent:true,opacity:.88});
      // Sport bonnet, roof spine, side glass and lower aero make AI opponents read as distinct cars.
      add(new THREE.BoxGeometry(1.2*widthScale,.055,1.05*lengthScale),accentMat,[0,1.205,.92*lengthScale]);
      add(new THREE.BoxGeometry(.58*widthScale,.035,.72*lengthScale),bodyMat,[0,1.735,-.08*lengthScale]);
      add(new THREE.BoxGeometry(.17,.035,2.9*lengthScale),stripeMat,[-.57*widthScale,1.23,-.02]);
      add(new THREE.BoxGeometry(.17,.035,2.9*lengthScale),stripeMat,[.57*widthScale,1.23,-.02]);
      for(const side of [-1,1]){
        add(new THREE.PlaneGeometry(.86*lengthScale,.28),glassMat,[side*.685*widthScale,1.43,-.12*lengthScale],[0,side*Math.PI/2,0]);
        add(new THREE.BoxGeometry(.075,.12,2.6*lengthScale),accentMat,[side*.925*widthScale,.72,0]);
        add(new THREE.BoxGeometry(.07,.045,2.7*lengthScale),stripeMat,[side*.94*widthScale,.84,0]);
      }
      add(new THREE.BoxGeometry(1.0,.08,.11),darkMat,[0,.91,2.075*lengthScale]);
      add(new THREE.BoxGeometry(1.85,.12,.25),trimMat,[0,.61,2.08*lengthScale]);
      add(new THREE.BoxGeometry(1.84,.12,.24),trimMat,[0,.61,-2.08*lengthScale]);
      for(const x of [-.66,.66]){
        add(new THREE.BoxGeometry(.39,.12,.075),glowMat,[x*widthScale,1.02,2.045*lengthScale]);
        add(new THREE.BoxGeometry(.44,.13,.08),redGlow,[x*widthScale,.96,-2.045*lengthScale]);
      }
      // A compact rear wing and diffuser give each rival a stronger performance-car silhouette.
      if(style>=2){
        add(new THREE.BoxGeometry(1.86,.075,.29),darkMat,[0,1.55,-1.78*lengthScale]);
        for(const x of [-.62,.62])add(new THREE.CylinderGeometry(.035,.055,.29,8),trimMat,[x*widthScale,1.40,-1.78*lengthScale]);
      }
      add(new THREE.BoxGeometry(1.12,.045,.28),darkMat,[0,.47,-1.91*lengthScale]);
      root.userData.powerupGlow=new THREE.Mesh(new THREE.TorusGeometry(1.55,.055,6,26),new THREE.MeshBasicMaterial({color:0x64efff,transparent:true,opacity:.76,depthWrite:false}));
      root.userData.powerupGlow.rotation.x=Math.PI/2;root.userData.powerupGlow.position.y=.17;root.userData.powerupGlow.visible=false;root.add(root.userData.powerupGlow);
    }
    const familyMat=makeMat(new THREE.Color(color).offsetHSL((design*0.047)%1,-.12,.18).getHex(),.28,.72);
    const familyDark=makeMat(0x101821,.3,.75);
    if(design===1){ // muscle: raised bonnet and twin hood louvres
      add(new THREE.BoxGeometry(.72*widthScale,.17,.82*lengthScale),familyDark,[0,1.22,1.05*lengthScale]);
      for(const x of [-.28,.28])add(new THREE.BoxGeometry(.08,.025,.58),familyMat,[x,1.315,.99*lengthScale]);
    }else if(design===2){ // hypercar: low side blades and rear diffuser
      for(const side of [-1,1])add(new THREE.BoxGeometry(.18,.15,2.55*lengthScale),familyMat,[side*1.0*widthScale,.69,0]);
      add(new THREE.BoxGeometry(1.48,.12,.42),familyDark,[0,.55,-2.05*lengthScale]);
    }else if(design===3){ // rally hatch: roof lights and raised wheel-arch hints
      add(new THREE.BoxGeometry(1.10,.10,.18),familyDark,[0,cabinHeight+.34,.48*lengthScale]);
      for(const x of [-.9,.9])add(new THREE.BoxGeometry(.16,.22,.72),familyMat,[x*widthScale,.83,1.15*lengthScale]);
    }else if(design===4){ // grand tourer: long bonnet spine and rear lip
      add(new THREE.BoxGeometry(.16,.035,1.46*lengthScale),familyMat,[0,1.215,.76*lengthScale]);
      add(new THREE.BoxGeometry(1.76,.09,.24),familyDark,[0,1.48,-1.82*lengthScale]);
    }else if(design===5){ // roadster: open-cockpit look and twin headrests
      add(new THREE.BoxGeometry(1.02,.08,.12),familyDark,[0,1.55,.55*lengthScale]);
      for(const x of [-.37,.37])add(new THREE.BoxGeometry(.27,.28,.24),familyMat,[x,1.72,-.62*lengthScale]);
    }else if(design===6){ // touring racer: oversized rear wing and door graphics
      add(new THREE.BoxGeometry(2.12,.10,.36),familyDark,[0,1.7,-1.82*lengthScale]);
      for(const side of [-1,1])add(new THREE.BoxGeometry(.055,.045,1.75*lengthScale),familyMat,[side*.96*widthScale,.93,.12]);
    }else if(design===7){ // electric concept: continuous light bar and roof spine
      add(new THREE.BoxGeometry(1.52,.075,.06),familyMat,[0,1.10,2.06*lengthScale]);
      add(new THREE.BoxGeometry(.24,.035,2.4*lengthScale),familyMat,[0,1.75,-.05]);
    }
    // Low-detail AI and traffic models carry the same per-car signature as the showroom version.
    const lowSignature=visualId%10,lowAccent=new THREE.Color(color).offsetHSL(((visualId*5)%19)/95-.1,-.08,.2).getHex(),lowAccentMat=makeMat(lowAccent,.28,.78);
    if(lowSignature===0)add(new THREE.BoxGeometry(.72*widthScale,.13,.48*lengthScale),familyDark,[0,1.19,1.22*lengthScale]);
    else if(lowSignature===1){for(const x of [-.48,.48])add(new THREE.BoxGeometry(.16,.035,2.35*lengthScale),lowAccentMat,[x*widthScale,1.19,.04]);}
    else if(lowSignature===2)add(new THREE.BoxGeometry(1.58*widthScale,.06,.11),lowAccentMat,[0,1.02,2.08*lengthScale]);
    else if(lowSignature===3)add(new THREE.BoxGeometry(.34*widthScale,.16,.62*lengthScale),familyDark,[0,1.84,-.12*lengthScale]);
    else if(lowSignature===4){for(const x of [-.38,0,.38])add(new THREE.BoxGeometry(.08,.025,.52*lengthScale),lowAccentMat,[x*widthScale,1.11,1.12*lengthScale]);}
    else if(lowSignature===5){for(const side of [-1,1])add(new THREE.BoxGeometry(.35,.06,.38),lowAccentMat,[side*1.0*widthScale,.52,1.72*lengthScale]);}
    else if(lowSignature===6){for(const side of [-1,1])add(new THREE.BoxGeometry(.085,.1,1.2*lengthScale),lowAccentMat,[side*.94*widthScale,.78,-.05]);}
    else if(lowSignature===7)add(new THREE.BoxGeometry(1.62*widthScale,.065,.07),lowAccentMat,[0,1.12,2.08*lengthScale]);
    else if(lowSignature===8)add(new THREE.BoxGeometry(1.74*widthScale,.075,.26),familyDark,[0,.5,-2.1*lengthScale]);
    else add(new THREE.BoxGeometry(.58*widthScale,.1,.68*lengthScale),lowAccentMat,[0,1.16,.92*lengthScale]);
    if(style===2)root.scale.set(1.02,1,1.06);else if(style===3)root.scale.set(.95,.99,.97);else if(style===4)root.scale.set(1.08,1.04,1.02);
    root.traverse((o,i)=>{if(o.isMesh){o.castShadow=i<8;o.receiveShadow=false;}});
    return root;
  }
  // Main body: beveled plan-view extrusion gives curved shoulders and tapered bumpers.
  const bodyGeo=new THREE.ExtrudeGeometry(planShape(.98*widthScale,2.22*lengthScale,-2.24*lengthScale,variantIndex,'body'),{depth:.58,bevelEnabled:true,bevelSegments:4,steps:1,bevelSize:.085,bevelThickness:.065,curveSegments:14});
  bodyGeo.rotateX(Math.PI/2);add(bodyGeo,bodyMat,[0,1.06,0]);
  // Lower sill and smooth hood / rear haunches add volume without blocky panels.
  add(new THREE.SphereGeometry(1,28,18),darkMat,[0,.58,0],null,[.91*widthScale,.22,2.12*lengthScale]);
  add(new THREE.SphereGeometry(1,28,18),bodyMat,[0,1.005,.99*lengthScale],null,[.78*widthScale,.115,.78*lengthScale]);
  add(new THREE.SphereGeometry(1,28,18),bodyMat,[0,1.0,-1.63*lengthScale],null,[.75*widthScale,.105,.48*lengthScale]);
  // Tapered glass cabin and roof are separate layers, so the car reads as a coupé rather than a box.
  const cabinGeo=new THREE.ExtrudeGeometry(planShape(.68*widthScale,.91*lengthScale,-1.08*lengthScale,variantIndex,'cabin'),{depth:.49,bevelEnabled:true,bevelSegments:4,steps:1,bevelSize:.055,bevelThickness:.045,curveSegments:12});
  cabinGeo.rotateX(Math.PI/2);add(cabinGeo,glassMat,[0,1.69,0]);
  const roofGeo=new THREE.ExtrudeGeometry(planShape(.53*widthScale,.48*lengthScale,-.66*lengthScale,(variantIndex+3)%20,'roof'),{depth:.055,bevelEnabled:true,bevelSegments:3,steps:1,bevelSize:.035,bevelThickness:.025,curveSegments:12});
  roofGeo.rotateX(Math.PI/2);add(roofGeo,bodyMat,[0,1.72,0]);
  // Angled windshield and rear glass catch highlights in the garage and on track.
  add(new THREE.PlaneGeometry(1.13*widthScale,.43),glassMat,[0,1.39,.78*lengthScale],[-.48,0,0]);
  add(new THREE.PlaneGeometry(1.08*widthScale,.38),glassMat,[0,1.38,-.94*lengthScale],[.48,Math.PI,0]);
  for(const side of [-1,1]){
    add(new THREE.PlaneGeometry(.96*lengthScale,.33),glassMat,[side*.675*widthScale,1.42,-.12*lengthScale],[0,side*Math.PI/2,0]);
    add(new THREE.SphereGeometry(.11,12,10),bodyMat,[side*.98*widthScale,1.24,.63*lengthScale],null,[1.25,.62,1.7]);
    // Side skirt and subtle wheel-arch highlights.
    add(new THREE.CapsuleGeometry(.075,2.05*lengthScale,.05,8),trimMat,[side*.94*widthScale,.48,0],[Math.PI/2,0,0]);
  }
  // Four detailed, animated wheel assemblies with metallic spokes and brake hubs.
  const wheelGeo=new THREE.CylinderGeometry(.43,.43,.31,30,1,false);
  const rimGeo=new THREE.CylinderGeometry(.275,.275,.325,28,1,false);
  const hubGeo=new THREE.CylinderGeometry(.105,.105,.34,18);
  for(const x of [-.91,.91])for(const z of [-1.38,1.38]){
    const wheel=add(wheelGeo,tireMat,[x*widthScale,.45,z*lengthScale],[0,0,Math.PI/2]);root.userData.wheels.push(wheel);if(z>0)root.userData.frontWheels.push(wheel);
    add(rimGeo,rimMat,[x*widthScale+(x>0?.17:-.17),.45,z*lengthScale],[0,0,Math.PI/2]);
    add(new THREE.TorusGeometry(.205,.035,8,24),makeMat(0x263b4b,.2,.95),[x*widthScale+(x>0?.19:-.19),.45,z*lengthScale],[0,Math.PI/2,0]);
    add(hubGeo,makeMat(0xd7e3ea,.2,.95),[x*widthScale+(x>0?.19:-.19),.45,z*lengthScale],[0,0,Math.PI/2]);
  }
  // Front fascia, grille, sculpted lamps and rear light bar.
  add(new THREE.SphereGeometry(1,22,12),darkMat,[0,.68,2.08*lengthScale],null,[.92*widthScale,.13,.19]);
  add(new THREE.BoxGeometry(.62,.12,.035),trimMat,[0,.76,2.265*lengthScale]);
  const headMat=new THREE.MeshStandardMaterial({color:0xeafaff,emissive:0xc8f4ff,emissiveIntensity:3.2,roughness:.18});
  const tailMat=new THREE.MeshStandardMaterial({color:0xff244f,emissive:0xff163d,emissiveIntensity:2.7,roughness:.22});
  for(const x of [-.69,.69]){
    add(new THREE.SphereGeometry(.18,18,14),headMat,[x*widthScale,.86,2.06*lengthScale],null,[1.3,.42,.22]);
    const tailLight=add(new THREE.SphereGeometry(.19,18,14),tailMat,[x*widthScale,.84,-2.13*lengthScale],null,[1.5,.4,.2]);root.userData.brakeLights.push(tailLight);
  }
  const headBeamMat=new THREE.MeshBasicMaterial({color:0xbceeff,transparent:true,opacity:.055,depthWrite:false,side:THREE.DoubleSide});
  for(const x of [-.62,.62])add(new THREE.ConeGeometry(.58,6.6,16,1,true),headBeamMat,[x*widthScale,.55,5.25*lengthScale],[Math.PI/2,0,0]);
  // Fine grille ribs and a sculpted lower front lip help the front end read clearly.
  for(let i=-2;i<=2;i++)add(new THREE.BoxGeometry(.055,.07,.035),makeMat(0x607280,.45,.72),[i*.19,.68,2.255*lengthScale]);
  add(new THREE.BoxGeometry(1.9,.075,.32),trimMat,[0,.48,2.22*lengthScale]);
  add(new THREE.BoxGeometry(1.8,.12,.28),trimMat,[0,.48,-2.15*lengthScale]);
  // Rank styling: aero grows with class, but the core silhouette stays coherent.
  const rank=modelSpec?.rank||null;
  if(['A','S','Exotic','Secret'].includes(rank)||style===4){
    add(new THREE.BoxGeometry(1.88,.07,.34),darkMat,[0,1.52,-1.78*lengthScale]);
    for(const x of [-.62,.62])add(new THREE.CylinderGeometry(.035,.055,.35,8),darkMat,[x*widthScale,1.34,-1.76*lengthScale]);
  }
  if(['S','Exotic','Secret'].includes(rank)){
    add(new THREE.BoxGeometry(2.05,.075,.36),makeMat(0x151d27,.22,.85),[0,1.9,-1.82*lengthScale]);
    for(const x of [-.73,.73])add(new THREE.CylinderGeometry(.035,.055,.42,8),darkMat,[x*widthScale,1.68,-1.8*lengthScale]);
  }
  // Distinct body-style details, blended from the earlier Midnight Run model variants.
  if(style===2){ // muscle: raised hood scoop and broad rear haunches
    add(new THREE.BoxGeometry(.58,.16,.72),darkMat,[0,1.12,1.13*lengthScale]);
    add(new THREE.BoxGeometry(1.86,.08,.2),trimMat,[0,.92,-1.7*lengthScale]);
    for(const x of [-.82,.82])add(new THREE.BoxGeometry(.1,.12,.72),darkMat,[x*widthScale,.93,1.38*lengthScale]);
  }else if(style===3){ // hyper: low nose canards and sculpted aero fins
    for(const side of [-1,1]){
      add(new THREE.BoxGeometry(.52,.07,.42),darkMat,[side*1.04*widthScale,.48,1.88*lengthScale],[0,side*.18,side*.12]);
      add(new THREE.BoxGeometry(.12,.28,1.22),trimMat,[side*.91*widthScale,.72,-.12*lengthScale],[0,side*.12,0]);
    }
    add(new THREE.BoxGeometry(1.35,.08,.24),darkMat,[0,.52,-2.28*lengthScale]);
  }else if(style===4){ // rally / tuner: roof intake and tall rear wing supports
    add(new THREE.BoxGeometry(.48,.17,.58),darkMat,[0,1.98,-.12*lengthScale]);
    for(const x of [-.72,.72])add(new THREE.CylinderGeometry(.045,.065,.48,8),darkMat,[x*widthScale,1.38,-1.82*lengthScale]);
    add(new THREE.BoxGeometry(2.02,.105,.32),darkMat,[0,1.62,-1.86*lengthScale]);
  }
  // Model-family bodywork: the 160 catalogue entries now have eight recognisable silhouettes.
  const familyAccent=new THREE.MeshStandardMaterial({color:new THREE.Color(color).offsetHSL((design*0.047)%1,-.1,.18),roughness:.26,metalness:.72,emissive:new THREE.Color(color).offsetHSL((design*0.047)%1,-.1,.08),emissiveIntensity:.12});
  if(design===0){ // compact street coupe
    add(new THREE.BoxGeometry(.12,.035,1.2*lengthScale),familyAccent,[0,1.145,.76*lengthScale]);
  }else if(design===1){ // wide muscle GT
    add(new THREE.BoxGeometry(.72,.17,.82*lengthScale),darkMat,[0,1.13,1.12*lengthScale]);
    for(const x of [-.42,.42])add(new THREE.BoxGeometry(.08,.025,.62),familyAccent,[x,.125+1.05,1.10*lengthScale]);
    add(new THREE.BoxGeometry(1.92,.10,.24),trimMat,[0,.66,-1.88*lengthScale]);
  }else if(design===2){ // low, wide hypercar
    for(const side of [-1,1]){add(new THREE.BoxGeometry(.22,.12,2.5*lengthScale),familyAccent,[side*1.02*widthScale,.65,0]);add(new THREE.BoxGeometry(.48,.06,.5),darkMat,[side*1.06*widthScale,.48,1.75*lengthScale],[0,side*.18,side*.14]);}
    add(new THREE.BoxGeometry(2.0,.09,.38),darkMat,[0,.5,-2.2*lengthScale]);
  }else if(design===3){ // rally / hot hatch
    add(new THREE.BoxGeometry(1.1,.12,.17),darkMat,[0,2.04,.32*lengthScale]);
    for(const x of [-.86,.86])add(new THREE.SphereGeometry(.42,12,8),familyAccent,[x*widthScale,.58,1.3*lengthScale],null,[.16,.32,.78]);
    add(new THREE.BoxGeometry(1.82,.12,.34),darkMat,[0,1.62,-1.85*lengthScale]);

// Release WebGL resources for cars that leave the scene so repeated races do not accumulate GPU memory.
function dispose3DObject(object,parentScene=scene){
  if(!object)return;
  parentScene?.remove(object);
  const geometries=new Set(),materials=new Set(),textures=new Set();
  object.traverse(child=>{
    if(child.geometry&&!geometries.has(child.geometry)){geometries.add(child.geometry);child.geometry.dispose();}
    const mats=Array.isArray(child.material)?child.material:(child.material?[child.material]:[]);
    mats.forEach(mat=>{
      if(!mat||materials.has(mat))return;materials.add(mat);
      ['map','alphaMap','aoMap','bumpMap','normalMap','roughnessMap','metalnessMap','emissiveMap'].forEach(key=>{
        const tex=mat[key];if(tex&&!textures.has(tex)){textures.add(tex);tex.dispose();}
      });
      mat.dispose();
    });
